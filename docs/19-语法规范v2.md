## 19.1 词法规则

```
identifier  ::= [A-Za-z_][A-Za-z0-9_]*                        # 决策 51：ASCII 唯一
integer     ::= "0x" hex+ | digit ( ["_"] digit )*             # 决策 38：hex/下划线
float       ::= digit ( ["_"] digit )* "." digit+ ( exponent )?
exponent    ::= ( "e" | "E" ) [ "+" | "-" ] digit+             # canonical 小写 e（formatter 管）
string      ::= '"' ( plain_char | escape | "${" expr "}" )* '"'   # 插值 = 拼串唯一形式
comment     ::= "//" 至行尾 | "///" 至行尾（文档注释，doctest 源）| "/*" … "*/"（不嵌套）
```

**词法卫生（决策 51）**：词法器拒绝一切 Bidi 方向控制字符（U+202A-202E、U+2066-2069、U+061C、U+200E/F）与零宽字符——包括字符串与注释内部，报错不警告（Trojan Source 免疫）。

**语句终止**：换行。行尾为二元运算符、逗号、`..`、未闭合括号/花括号时语句延续——确定性规则，无 ASI 式猜测。`${}` 插值内禁止换行。无缩进语义。

**记号一义性**：前缀 `!` = 逻辑非 / 后缀 `!` = Option 解包（位置分离）；`|` = 按位或与 tag 变体分隔（声明/表达式位置分离）；`..` = 区间与字段展开（表达式/struct 字面量位置分离）——三处均为**位置分离的同 token**，无歧义、不新增关键字。

## 19.2 语法规则（EBNF v2，56 条产式）

```
program       ::= { top_decl }
top_decl      ::= [ "pub" ] ( import_decl | type_decl | fn_decl | struct_decl
                | tag_decl | protocol_decl | impl_decl | const_decl )      # 决策 32/50

import_decl   ::= "import" identifier [ "as" identifier ] "from" string    # 具名导入+可选别名
type_decl     ::= "type" identifier [ type_params ] "=" ( type | "distinct" type )
const_decl    ::= "const" identifier ":" type "=" expr                      # 编译期可折叠（决策 50）

fn_decl       ::= "fn" identifier [ type_params ] "(" [ param { "," param } ] ")" [ "->" type ] block   # 决策 84：体单形式
type_params   ::= "<" identifier { "," identifier } [ ":" protocol_ref ] ">"  # 决策 49：T: Ord；决策 85：`<T>` 唯一
param         ::= identifier ":" type                                        # 无默认/命名参数（48）

struct_decl   ::= "struct" identifier [ type_params ] "{" field_decl { newline } "}"   # 递归类型允许（49）
field_decl    ::= identifier ":" type
tag_decl      ::= "tag" identifier [ type_params ] "=" variant { "|" variant }
variant       ::= identifier [ "(" type ")" ]

protocol_decl ::= "protocol" identifier [ type_params ] "{" { protocol_fn } "}"
protocol_fn   ::= "fn" identifier "(" [ param { "," param } ] ")" [ "->" type ]
impl_decl     ::= "impl" protocol_ref "for" identifier [ type_params ] "{" { fn_decl } "}"

block         ::= "{" { stmt } "}"
stmt          ::= let_decl | expr | while_stmt | for_stmt | return_stmt
                | throw_stmt | break_stmt | continue_stmt | block            # 决策 47
let_decl      ::= "let" [ "mut" ] identifier [ ":" type ] "=" expr
while_stmt    ::= "while" expr block
for_stmt      ::= "for" identifier "in" expr block
return_stmt   ::= "return" [ expr ]
throw_stmt    ::= "throw" expr
break_stmt    ::= "break"                                                    # 无标签（47）
continue_stmt ::= "continue"

expr          ::= assign_expr
assign_expr   ::= lvalue ( "=" | "+=" | "-=" ) expr | or_expr
lvalue        ::= identifier { "." identifier | "[" expr "]" }                # 决策 41：owned mut 字段赋值
or_expr       ::= range_expr { ( "||" | "or" ) range_expr }                    # 决策 RW-c：or 软关键字
range_expr    ::= and_expr [ ".." and_expr ]                                  # 决策 42：半开，非结合
and_expr      ::= pipe_expr { ( "&&" | "and" ) pipe_expr }                    # 决策 RW-c：and 软关键字
pipe_expr     ::= cmp_expr { "|>" cmp_expr }                                  # 决策 25：左线程
cmp_expr      ::= bitor_expr { ( "==" | "!=" | "<" | ">" | "<=" | ">=" ) bitor_expr }  # 非结合
bitor_expr    ::= bitxor_expr { "|" bitxor_expr }
bitxor_expr   ::= bitand_expr { "^" bitand_expr }
bitand_expr   ::= shift_expr { "&" shift_expr }
shift_expr    ::= add_expr { ( "<<" | ">>" ) add_expr }
add_expr      ::= mul_expr { ( "+" | "-" ) mul_expr }
mul_expr      ::= unary { ( "*" | "/" | "%" ) unary }                          # 决策 44：floor 语义
unary         ::= ( "!" | "-" ) unary | postfix
postfix       ::= primary { "(" args ")" | "." identifier | index_suffix
                | "!" | "as" type }                                          # 后缀 ! 解包；as 唯一转换（26/23，RW-b）
index_suffix  ::= "[" expr "]" | "[" expr ".." expr "]"                        # 单元素/切片（42）
args          ::= expr { "," expr }

primary       ::= literal | identifier | lambda | if_expr | match_expr
                | try_expr | await_expr | struct_lit | list_lit | map_lit | group
lambda        ::= "fn" "(" [ lam_param { "," lam_param } ] ")" ( "=" expr | block )   # 决策 84：`= expr` 表达式体仅属 lambda
lam_param     ::= identifier [ ":" type ]                                     # 顶层 fn 必标、lambda 可推断
list_lit      ::= "[" [ expr { "," expr } ] "]"
map_lit       ::= "[" [ expr ":" expr { "," expr ":" expr } ] "]"             # 空字面量须标注（52）
if_expr       ::= "if" expr block "else" ( if_expr | block )                   # 作值必带 else
                | "if" expr block                                              # 语句位可省 else
match_expr    ::= "match" expr "{" { match_arm } "}"
match_arm     ::= pattern "->" ( expr | block )
pattern       ::= literal | identifier | variant_pattern | "_"                 # 无解构模式（嵌套/字面量载荷见决策 86 注）
variant_pattern ::= identifier [ "(" ( identifier | "_" ) ")" ]                # 决策 86：`Node(_)` 显式丢弃载荷
try_expr      ::= "try" expr                                                   # 决策 54：双传播
                | "try" expr "catch" "{" { match_arm } "}"                     # 决策 87：可捕获失败通道（catch 与 try 体同行；臂结果须与 try 体类型统一，错误值类型任意 → 必有兜底臂）
await_expr    ::= "await" expr                                                 # 仅宿主能力
struct_lit    ::= identifier [ type_args ] "{" [ field_init { "," field_init } ] [ ".." expr ] "}"
field_init    ::= identifier ":" expr
type_args     ::= "<" type { "," type } ">"
group         ::= "(" expr ")"
literal       ::= integer | float | string | "true" | "false" | "nil"
```

### 类型文法 v2（与表达式文法严格分离）

```
type      ::= prim | named | option | result | list | map | fn_type | "Json"
prim      ::= "int" | "float" | "bool" | "string" | "rune" | "unit"            # 决策 46：rune
named     ::= identifier [ type_args ]
option    ::= type "?"                                                          # 决策 39：T?? 拒绝
result    ::= "Result" "<" type "," type ">"
list      ::= "[" type "]"
map       ::= "[" type ":" type "]"                                             # K 约束 Hash+Ord（30）
fn_type   ::= "(" [ type { "," type } ] ")" "->" type
```

**相对 v1 的文法删改**：**删** `record`（决策 53，数据声明一律 struct）；**增** `pub` 前缀、`break/continue`、`rune`、`type_params`（泛型类型）、`..` 区间表达式、`..` 字段展开、切片下标、list/map 字面量、import 可选别名、lvalue 字段/索引赋值、lambda 参数类型可省。
**集合 API 约定**：`len/map/filter/fold/sort` 等全部为**自由函数、数据第一参数**（与 `|>` 左线程配套，决策 25）——内建集合无方法调用，`.field` 后缀仅用于 struct 字段。

## 19.3 运算符优先级表（低 → 高，非结合标 ※）

| 级 | 运算符 | 结合 | 备注 |
|---|---|---|---|
| 1 | `..` | ※ 非结合 | 仅 int 操作数；切片与 for 共用（决策 42） |
| 2 | `\|>` | 左 | 左线程：`x \|> f(y)` = `f(x, y)`；不可重载 |
| 3 | `||` | 左 | |
| 4 | `&&` | 左 | |
| 5 | `== != < > <= >=` | ※ 非结合 | `a == b == c` 是类型错误（bool≠比较对象） |
| 6-9 | `\| ^ & << >>` | 左 | 移位越界抛错（37） |
| 10 | `+ -` | 左 | int/float 严格隔离（26）；溢出抛错（29） |
| 11 | `* / %` | 左 | floor 除/模（44）；溢出抛错 |
| 12 | `! -`（前缀） | 右 | |
| 13 | `. [] () !`（后缀） | 左 | 后缀 `!` 解包 |

## 19.5 完整示例（覆盖全部表面形式的唯一权威样例）

```
import math from "std"                       // 具名导入，限定访问 math.sqrt
import json as j from "std"                  // 可选别名
import string from "std"                     // 决策 43：import 必用（RW-h）

pub struct Vec3 {                            // pub = 对外可见
    x: float
    y: float
    z: float
}

impl Add for Vec3 {                          // 协议显式 impl（决策 21）
    fn add(a: Vec3, b: Vec3) -> Vec3 {
        return Vec3 { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z }
    }
}

type EntityId = distinct int                 // 决策 23（Jai `HandleA :: u32` 独立收敛到同设计——外部验证，23 号 J8）
struct Node2<T> {                            // 多载荷变体走 struct（无元组，Tier C）
    left: Tree<T>?
    val: T
}
type Tree<T> = Node(Node2<T>) | Leaf         // 泛型 + 递归（决策 49）

pub tag Err = NotFound | BadInput(string) | TooDeep

const MAX_DEPTH: int = 0xFF                  // 决策 38：hex 字面量，编译期折叠

fn depth(t: Tree[int], d: int) -> Result<int, Err> {
    if d > MAX_DEPTH {
        return Err(TooDeep)
    }
    return match t {                         // return 唯一返回形式（RW-g）
        Leaf -> Ok(d)
        Node(p) -> Ok(try depth(p.left!, d + 1))     // try 双传播（决策 54），Ok 包回统一臂类型
    }
}

pub fn summarize(ctx: Ctx, items: [Item]) -> Result<string, Err> {
    let mut best: Item? = nil                // 局部推断，永不写类型
    let total = items |> map(fn(it) = it.size) |> fold(0, fn(a, b) = a + b)
    for it in items {                        // 唯一遍历循环
        if it.size >= 1_000_000 {            // 决策 38：下划线分隔
            continue
        }
        if it.size == 0 {
            break                            // 无标签（决策 47）
        }
        if best == nil or it.size > best!.size {
            best = it                        // T→T? 隐式加宽（决策 27）
        }
    }
    if best != nil {
        let b = best                         // 分支内 b: Item（决策 40 智能收窄）
        let renamed = Item { ..b, title: "picked" }  // 函数式字段更新（决策 41）
        await ctx.log("${renamed.title} of ${len(items)}")   // 插值唯一拼串
    }
    let mut m: [string: int] = [:]           // 空 map 须标注（决策 52）
    m["a"] = 1                               // owned mut 索引赋值
    let n = m["b"]                           // n: int?（决策 55）
    let head = items[0..2]                   // 半开切片（决策 42）
    let first = string.to_runes("你好")[0]   // rune 经迭代/切片获取，禁 s[i]（46）
    let wrapped = (total as int) % len(items)        // floor-mod 恒非负（决策 44）
    if n == nil {
        return Err(NotFound)                 // 失败路径用 return Err（throw 是穿透 Raise，26 号 D-21；决策 87 后可在 catch 作用域内被本地截获）
    }
    return Ok("head=${len(head)} first=${first} mod=${wrapped}")
}

fn norm(v: Vec3) -> float {                  // 消费 math：import 必用（决策 43，RW-h）
    return math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z)
}

fn config(s: string) -> Result<int, Err> {   // 消费 j 别名：import as 示范
    let n = parse_int(s)                     // 裸内建（更正 2026-10-12：parse_int/parse_float
    if n == nil {                            //   不在 json 模块，26 号 D-19；j 模块仅 parse/stringify）
        throw BadInput(s)
    }
    return Ok(n)                             // n 已 nil 收窄为 int（决策 40）
}
```

## 19.7 关键字冻结表（29 个，+pub/break/continue/rune/while/catch）

`let mut fn return if else for while in match struct tag protocol impl type import const as try catch throw await nil true false distinct pub break continue rune`

冻结条款不变：新增关键字须连带删除或合并一个现有表达方式，总量只减不增。例外记账：`catch` 为决策 87（错误模型收敛，32 号）新增——该决策同时砍掉「throw 不可恢复」这一整条失败语义（原语言最激进的角落），净表达力为减。`or`/`and` **不是关键字**——软关键字（仅在运算符位按 `||`/`&&` 识别，RW-c）。
